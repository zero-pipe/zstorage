set(ZCS_DEMO_SERVER_RUNTIME demo_server_runtime.c)

function(zcs_add_demo name)
    add_executable(${name} ${name}.c)
    # ZCS is static on MSVC: executables must link znet for symbols used inside ZCS (e.g. zn_info)
    target_link_libraries(${name} PRIVATE zcs znet)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()

function(zcs_add_server_demo name)
    add_executable(${name} ${name}.c ${ZCS_DEMO_SERVER_RUNTIME})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${name} PRIVATE zcs znet)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()

zcs_add_server_demo(demo_media_server)

if (ZCS_HAVE_SRT AND WIN32)
    zcs_copy_srt_dll(demo_media_server)
endif()
