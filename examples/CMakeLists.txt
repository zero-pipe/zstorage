set(ZCS_SERVER_RUNTIME demo_server_runtime.c)

add_executable(zcs-cli zcs.c ${ZCS_SERVER_RUNTIME})
target_include_directories(zcs-cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# ZCS is static on MSVC: executables must link znet for symbols used inside ZCS (e.g. zn_info)
target_link_libraries(zcs-cli PRIVATE zcs znet)
set_target_properties(zcs-cli PROPERTIES OUTPUT_NAME zcs)
if(MSVC)
    target_compile_definitions(zcs-cli PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
